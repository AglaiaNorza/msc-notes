\chapter{Computational Complexity (Cheat Sheet)}

\section{P and NP}
\begin{defframe}{The class NP}{}
There are two ways to define the complexity class NP:
\begin{enumerate}
    \item NP is the set of decision problems \textit{solvable} in polynomial time by a \textit{nondeterministic} Turing machine
    \item NP is the set of decision problems \textit{verifiable} in polynomial time by a \textit{deterministic} Turing machine:
        \begin{itemize}
            \item that is, $L \in \text{NP}$ if there are a polynomial $p$ and a polynomial-time verifier $V$ such that for every $x$:
        $$x \in L \iff \exists u \text{ with } |u|\leq p(|x|) \text{ s.t. } V(x, u)=\text{accept}$$
        \end{itemize}
\end{enumerate}
\end{defframe}

\begin{cframe}{ColorLemma}{Showing $L \in$ NP}
    \vspace{-1em}
    \begin{enumerate}
        \item define the certificate $u$ (the proposed solution)
        \item show that $|u| \leq $ a polynomial in the input length $|x|$ (NB: remember to count bits - $n$ vertices take up $n \log n$ bits, not $n$)
        \item explain the mechanism of the verifier $V(x, u)$
        \item confirm that the running time of the verifier is polynomial in $|x|$
        \item prove both directions of correctness:
            \begin{itemize}
                \item explain why the verifier accepts a YES-instance
                \item explain why the verifier rejects a NO-instance (it cannot loop !)

                    or, assume that the verifier accepts some certificate $u$, and prove that $x$ must be a YES-instance
            \end{itemize}
    \end{enumerate}
\end{cframe}

\begin{defframe}{Karp Reduction}{}
    A polynomial-time Karp reduction translates an instance of problem $A$ into an instance of problem $B$. Upper bounds transfer from $B$ to $A$, meaning if $B \in P$, then $A \in P$. Conversely, hardness transfers from $A$ to $B$ (if $A \leq_{p} B $ and $A \not\in P$, $B \not\in P$). 

    Reductions are also transitive, meaning if $A \leq_{p}B $ and $B \leq_{p} C$, then $A \leq_{p} C$.

We have that $A \leq_{p} B$ iff there is a polynomial function $f$ such that, for every string $x$:
$$x \in A \iff f(x) \in B$$
\end{defframe}

\begin{cframe}{ColorLemma}{Showing $A \leq_{p} B$}
    \begin{enumerate}
        \item define the function $f(\text{instance of } A) = \text{instance of } B$, which must be computed from the instance alone
        \item state the size of $f(x)$ and the cost of computing it, in terms of $|x|$
        \item correctness ($\implies $): prove that if $x$ is a YES-instance of $A$, then $f(x)$ is a YES-instance of $B$ (typically involves constructing a solution for $f(x)$ from a solution of $x$)
        \item correctness ($\impliedby $): prove that if $f(x)$ is a YES-instance of $B$, then $x$ is a YES-instance of $A$ (typically involves recovering a solution for $x$ from a solution of $f(x)$)
            
            or, do the contrapositive: if $x$ is a NO-instance of $A$, then $f(x)$ is a NO-instance of $B$
    \end{enumerate}
\end{cframe}

The core reductions covered in the course are:
\begin{itemize}
\item \textbf{Logic to logic:} $\text{SAT} \le_p \text{NAE-SAT}$
    \begin{itemize}
        \item \textit{Reduction by symmetry breaking:} pick a fresh variable $s$ and append it to every clause: $(\ell_1 \lor \dots \lor \ell_r) \mapsto (\ell_1, \dots, \ell_r, s)$.
        \begin{itemize}
            \item NAE has flip symmetry: if $\beta$ satisfies, flipping all values $\beta'$ also satisfies.
            \item the shared $s$ acts as a reference anchor: without loss of generality, take the solution where $s = \text{false}$.
            \item since $s$ guarantees the required false in every clause, the clause is NAE-satisfied if and only if at least one original literal $\ell_i$ is true (which is the standard SAT condition)
        \end{itemize}
    \end{itemize}
\item \textbf{logic to graphs:} $\text{3-SAT} \le_p \text{CLIQUE}$
    \begin{itemize}
        \item for a formula with $m$ clauses:
        \begin{itemize}
            \item create 3 vertices per clause (one per literal occurrence), put an edge between two vertices iff they are in different clauses and consistent ($u \neq \neg v$), set target clique size $k = m$
            \item since there are no edges within the same clause, a clique of size $m$ must pick exactly one literal per clause; because edges only connect non-contradictory literals, picking an $m$-clique never selects both $x_i$ and $\neg x_i$, directly defining a valid satisfying truth assignment
        \end{itemize}
    \end{itemize}
\item \textbf{graph to graph:} $\text{CLIQUE} \le_p \text{INDEPENDENT SET} \le_p \text{VERTEX COVER}$

    \begin{itemize}
        \item $\text{CLIQUE} \le_p \text{IS}$:
        \begin{itemize}
            \item \textit{map:} $(G, k) \mapsto (\overline{G}, k)$, where $\overline{G}$ is the complement graph
            \item a subset $S \subseteq V$ forms a clique in $G$ (all pairs connected) $\iff$ $S$ is an independent set in $\overline{G}$ (no pairs connected)
        \end{itemize}
        \item $\text{IS} \le_p \text{VC}$:
        \begin{itemize}
            \item \textit{map:} $(G, k) \mapsto (G, |V| - k)$ (flip the target size)
            \item $S$ is an independent set $\iff V \setminus S$ is a vertex cover: since no edge can have both endpoints in $S$, every edge must touch at least one vertex in $V \setminus S$; thus, an IS of size $\ge k$ exists $\iff$ a VC of size $\le |V| - k$ exists
        \end{itemize}
    \end{itemize}

\item \textbf{Logic to Coloring:} $\text{NAE-3-SAT} \le_p \text{3-COLORING}$
    \begin{itemize}
        \item \textit{Construction / Gadgets:}
        \begin{itemize}
            \item have a neutral vertex $B$ (assigned color \textit{base}; remaining two colors represent \textit{true} and \textit{false}).
            \item for each variable $x$, connect $x$ and $\neg x$ to each other and both to $B$ (forming a triangle with $B$); for each clause $(\ell_1, \ell_2, \ell_3)$, create a fresh triangle $u_1, u_2, u_3$ and add edges $(u_1, \ell_1)$, $(u_2, \ell_2)$, and $(u_3, \ell_3)$
            \item connecting every literal to $B$ forces all literals to take colors from $\{\text{true}, \text{false}\}$, with $x$ and $\neg x$ having opposite colors.
            \item the triangle $\{u_1, u_2, u_3\}$ requires all 3 colors.
            \begin{itemize}
                \item if $\ell_1, \ell_2, \ell_3$ are all equal (all true or all false), that color is blocked for all $u_i$, leaving only 2 available colors for a 3-cycle (impossible to 3-color).
                \item if $\ell_1, \ell_2, \ell_3$ are not all equal, the three colors $\{\text{true}, \text{false}, B\}$ can always be validly assigned to $\{u_1, u_2, u_3\}$.
            \end{itemize}
        \end{itemize}
    \end{itemize}

\end{itemize}

\begin{defframe}{Gadgets}{}
Often, pieces of the target problem (e.g., a subgraph) can be designed to simulate a specific component of the source problem (e.g., a variable or a clause)
\begin{enumerate}
    \item \textbf{choice gadget:} how do you represent a boolean variable making a choice (True/False)?
    \begin{itemize}
        \item \textit{example:} in mapping to $\text{3-COLORING}$, a variable $x$ is represented by a pair of adjacent vertices ($x$ and $\neg x$) that are both connected to a universal "Base" vertex; the Base takes one color, forcing $x$ and $\neg x$ to take the remaining two colors (representing True and False)
    \end{itemize}
    \item \textbf{constraint gadget:} how do you enforce the rules of the source problem (e.g., "at least one literal must be true")?
    \begin{itemize}
        \item \textit{example:} build a small subgraph connecting the literal vertices of a clause to a target vertex; the subgraph's edges are designed so that the whole gadget is only valid (e.g., fully colorable) if the inputs satisfy the source problem's specific rule
    \end{itemize}
\end{enumerate}
\end{defframe}

\section{Space Classes (L and NL)}

\begin{defframe}{The Classes L and NL}{}
Instead of bounding time, we bound the memory (space) used on the work tape. The input tape is read-only, and we only count the memory used on the read/write work tape.

\begin{itemize}
    \item \textbf{L (log-space):} the set of decision problems solvable by a \textit{deterministic} Turing machine using $O(\log|x|)$ space
    \item \textbf{NL (nondeterministic log-space):} the set of decision problems solvable by a \textit{nondeterministic} Turing machine using $O(\log|x|)$ space
\end{itemize}
\textbf{Savitch / Immerman-Szelepcsényi:} $\text{NL} = \text{coNL}$.
\end{defframe}

\begin{cframe}{ColorLemma}{Showing $L \in \text{L}$ or $\text{NL}$}
To show a problem is in L or NL, define an algorithm and prove it uses logarithmic space
    \begin{enumerate}
        \item write the algorithm as a numbered list.
        \item list exactly what the work tape needs to hold (e.g., counters, vertex indices, pointers).
        \item write out the exact bit size for each variable, and show that the sum of all bit sizes is bounded by $O(\log|x|)$
        \begin{itemize}
            \item \textit{Note:} a vertex index out of $n$ vertices takes $\lceil \log_2 n \rceil$ bits; a counter up to $n$ takes $\lceil \log_2 n \rceil$ bits
        \end{itemize}
    \end{enumerate}
\end{cframe}

\begin{defframe}{Log-space Reductions ($A \leq_{L} B$)}{}
A log-space reduction from $A$ to $B$ is a deterministic algorithm that computes a mapping function $f$, transforming an instance of $A$ into an instance of $B$, using only $O(\log|x|)$ work-tape space. 
\end{defframe}

\begin{cframe}{ColorLemma}{Showing $A \leq_{L} B$}
    \vspace{-1em}
    \begin{enumerate}
        \item define $f(\text{instance of } A) = \text{instance of } B$, computed deterministically.
        \item state the variables needed to compute the map and their bit sizes, showing the sum is $O(\log|x|)$. (Do not count the space used to write the output, only the work tape).
        \item correctness ($\implies$): prove that if $x$ is a YES-instance of $A$, then $f(x)$ is a YES-instance of $B$
    \item crrectness ($\impliedby$): prove that if $f(x)$ is a YES-instance of $B$, then $x$ is a YES-instance of $A$ (or prove the contrapositive for NO-instances)
    \end{enumerate}
\end{cframe}

\paragraph{Problems in L}
\begin{itemize}
    \item \textbf{BALANCED PARENTHESES:} given a string over '(' and ')', are they properly balanced? (uses a counter to track the current depth).
    \item \textbf{PALINDROME:} given a string, is it equal to its reverse? (uses two pointers)
    \item \textbf{EVEN-DEGREE:} given an undirected graph, does every vertex have an even degree? (uses a counter to iterate over edges for each vertex)
    \item \textbf{UNDIRECTED PATH:} given an undirected graph and vertices $s, t$, is there a path from $s$ to $t$? (the directed version is NL-complete !)
\end{itemize}


\paragraph{Problems in NL}
\begin{itemize}
    \item \textbf{PATH (NL-Complete):} given a directed graph and vertices $s, t$, is there a directed path from $s$ to $t$? \textit{(canonical NL-complete problem; NL-hardness proofs start by reducing from PATH, often using configuration graphs)}
    \item \textbf{STRONG-CONNECTIVITY:} given a directed graph, is there a directed path between every pair of vertices?
    \item \textbf{CYCLE-THROUGH:} given a directed graph and a vertex $v$, is there a directed cycle containing $v$?
    \item \textbf{2-SAT and 2-UNSAT:} is a given 2-CNF formula satisfiable/unsatisfiable? (building an implication graph and searching for a directed path from a literal $x$ to $\neg x$ and vice versa)
\end{itemize}

\section{The polynomial hierarchy}
The polynomial hierarchy (PH) generalizes the complexity classes P, NP and coNP by classifying computational problems based on the number of alternating quantifiers required to verify their solutions.

\textcolor{gray}{(In standard NP, a non-deterministic machine guesses one certificate and checks it in polynomial time. The polynomial hierarchy extends this by allowing multiple, alternating phases of guessing. Each phase represents a shift in the logical burden of proof - from "does there exist?" ($\exists$) to "is it true for all?" ($\forall$). The number of times this burden shifts dictates the problem's level in the hierarchy.)}

The base of every problem in the polynomial hierarchy is the polynomial-time predicate $R$

\begin{defframe}{Polynomial-time predicate}{}
A \textbf{polynomial-time predicate} $R(x, u_{1}, \dots, u_{k})$ is a condition on strings that a polynomial-time algorithm checks.
\begin{itemize}
    \item $u_{i}$ are the certificates (guesses or moves generated by the quantifiers)
    \item the length of every $u_{i}$ needs to be bounded by some polynomial relative to the size of the input $x$ ($|u_{i}| \leq p(|x|)$)
    \item $R$ then evaluates the conditions and outputs true/false in polynomial time
\end{itemize}
\end{defframe}

The classes of PH depend on the quantifier structure of the problem:

\begin{defframe}{Quantifier block}{}
    A block is a \textit{maximal run of equal quantifiers}. Two quantifiers of the same kind in a row are one block, since a pair of strings is one string; the \textit{quantifier level} counts blocks.
\end{defframe}

\begin{cframe}{ColorLemma}{The class $\Sigma_{k}$}{}
    A language $L$ belongs in the class $\Sigma_{k}$ if its defining logical statement starts with an existential quantifier, and has exactly $k-1$ alternations.
    \begin{itemize}
        \item e.g. $x \in L \in \Sigma_{3} \iff \exists u_1 \forall u_2 \exists u_3 : R(x, u_1, u_2, u_3)$.
    \end{itemize}
\end{cframe}

\begin{itemize}
    \item We have that $\Sigma_{1} =$ NP !
\end{itemize}

\begin{cframe}{ColorThm}{The class $\Pi_{k}$}{}
A language $L$ belongs in the class $\Pi_{k}$ if its defining logical statement starts with a universal quantifier, and has exactly $k-1$ alternations.
    \begin{itemize}
        \item e.g. $x \in L \in \Pi_{2} \iff \forall u_1 \exists u_2 : R(x, u_1, u_2)$.
    \end{itemize}

$\Pi_{k}$ is the complement of $\Sigma_{k}$): $\Pi_{i} = \{L : \bar{L} \in \Sigma_{i}\}$ (negating the satatement swaps every $\exists  $ with a $\forall  $)
\end{cframe}

\begin{itemize}
    \item We have that $\Pi_{1} =$ coNP !
\end{itemize}

\subsection{The Hierarchy Ladder}
We have that:
$$\Sigma_i \cup \Pi_i \subseteq \Sigma_{i+1} \cap \Pi_{i+1}$$
, which means that any language in $\Sigma_i$ or $\Pi_i$ is also contained within both classes of the next level up ($\Sigma_{i+1}$ and $\Pi_{i+1}$). This is proven by adding a ``dummy'' quantifier.

If a language is defined by a polynomial-time predicate $R$, we can artificially introduce a new quantifier over a variable $v$ that $R$ completely ignores. Because the verifier never evaluates $v$, the truth value of the statement is unchanged, but the formula gains the extra alternation required to place it in a higher class.

\begin{itemize}
\item moving from $\Sigma_1$ to $\Sigma_2$ ($\exists \forall$): 
    \subitem suppose $L \in \Sigma_1$ (NP), defined as $x \in L \iff \exists u : R(x, u)$. 
    
    we can rewrite this as $x \in L \iff \exists u \forall v : R'(x, u, v)$, where $R'$ executes $R(x,u)$ and ignores $v$ - the problem is now technically in $\Sigma_2$
    
\item moving from $\Sigma_1$ to $\Pi_2$ ($\forall \exists$):
    \subitem using the same $L \in \Sigma_1$, we can alternatively pad the front of the statement: $x \in L \iff \forall v \exists u : R'(x, u, v)$, where $R'$ again ignores $v$ - the problem is now technically in $\Pi_2$.
\end{itemize}

Through this padding mechanism, the lower classes are subsumed by the intersection of the higher classes.

\subsection{The Quantifier Form and Absorption Rule}

\begin{cframe}{ColorLemma}{The Quantifier Form}
    To define a language $L$ using the quantifier form, it is necessary to provide the statement, sizes, predicate, and class: 
    \begin{enumerate}
        \item \textbf{statement}: explicitly state $x \in L \iff Q_1 u$ (what $u$ is, its size) $Q_2 v$ (what $v$ is, its size): $R(x, u, v)$, written in words (e.g., ``there is a \dots such that for every \dots'')
        \item \textbf{sizes}: state that each object is bounded by an explicit polynomial in $|x|$\\
        \textcolor{gray}{(N.B: a condition that is part of the question, e.g. ``the set has at least $k$ vertices'', is checked inside $R$; it is not a size bound.)}
        \item \textbf{predicate}: state that $R$ is decidable in polynomial time, providing the reason
        \item \textbf{class}: read off the prefix: $\exists \to \text{NP}$, $\forall \to \text{coNP}$, $\exists\forall \to \Sigma_2$, $\forall\exists \to \Pi_2$
    \end{enumerate}
\end{cframe}


Not every $\forall $ or $\exists  $ statement creates a new hierarchy level:
\begin{cframe}{ColorThm}{The absorb rule}
The absorb rule dictates that if the candidate space for a quantifier is small enough that it can be listed and checked in polynomial time (e.g., checking every node in a graph), that quantifier is "absorbed" directly into the polynomial-time predicate $R$ (and it is called a \textbf{loop}). 

A quantifier only remains in the prefix - and thus pushes the problem to a higher complexity class - if it iterates over an exponentially large candidate space, e.g. all possible $2^n$ Boolean truth assignments 
\end{cframe}

\paragraph{Example: INDEPENDENT SET.}
Consider the problem: \textit{is there a set $S \subseteq V$ with $|S| \ge k$, such that there are no edges inside $S$?}

The logical statement is: \textit{there exists} ($\exists$) a set $S \subseteq V$ with $|S| \ge k$ such that \textit{for every pair} ($\forall$) $\{a, b\} \subseteq S$, $ab \notin E$.

\begin{itemize}
    \item The $\exists$ block remains: thge candidate space for $S$ is $2^n$ (all possible subsets of $n$ vertices). Because this scales exponentially, a deterministic machine cannot check all possibilities in polynomial time: the quantifier block remains.
    \item The $\forall$ block is absorbed: once a specific candidate set $S$ is handed to the verifier, the inner condition checks ``for every pair''. The maximum number of vertex pairs in $S$ is bounded by $n^2$. Therefore, the machine simply runs a nested loop over the pairs to check the adjacency list. The $\forall$ quantifier is \textbf{absorbed} into the deterministic polynomial-time predicate $R$.
\end{itemize}

Therefore, the problem is in $\Sigma_1 = $ NP.


To understand complete problems for the polynomial hierarchy, we use Quantified Boolean Formulas, which are the natural extension of the standard Boolean satisfiability problem (SAT).

\begin{defframe}{}{}
A quantified Boolean formula is typically written in prenex form (meaning all quantifiers are pulled to the front) followed by a quantifier-free Boolean formula called the matrix ($\psi$).
$$Q_1x_1 Q_2x_2 \dots Q_qx_q [\psi]$$
\begin{itemize}
    \item where each $Q_{i}$ is a quantifier ($\exists  $ or $\forall  $)
\end{itemize}

The formula is fully quantified, meaning every variable in the matrix $\psi$ is bound by a quantifier (so the entire formula is either true or false).
\end{defframe}

Just as SAT is complete for NP, specific formats of QBFs act as the canonical complete problems for every level of the hierarchy.

\begin{itemize}
    \item $\Sigma_2\text{-SAT}$: "given a formula $\exists X \forall Y [\psi]$, is it true?" is a complete problem for $\Sigma_2$. 
        \subitem the matrix $\psi$ must be an unrestricted Boolean formula: if $\psi$ were in CNF, the $\forall Y$ check could be evaluated clause by clause, and the complexity would drop down to NP
    \item $\Pi_2\text{-SAT}$: "given a formula $\forall X \exists Y [\psi]$, is it true?" is $\Pi_2$-complete, and it remains complete even if $\psi$ is in CNF
\end{itemize}

\subsection{The class DP}
The class DP sits just above NP and coNP, but below the second level of the polynomial hierarchy ($\Sigma_2 \cap \Pi_2$). It captures problems that require solving an NP question and a coNP question simultaneously.

\begin{defframe}{DP}{}
    A language belongs to DP if it is the intersection of two languages over the same instances: one from NP, and one from coNP:
 $$\text{DP} = \{L_1 \cap L_2 : L_1 \in \text{NP}, L_2 \in \text{coNP}\}$$
\end{defframe}

N.B.: the class DP is different from NP $\cap $ coNP, which represents the class of single languages that have both a fast yes and a fast no verifier
\begin{itemize}
    \item actually, DP = NP $\cap$ coNP if and only if NP = coNP
\end{itemize}

\begin{cframe}{ColorLemma}{Showing $L \in$ DP}
    To show that a language $L$ is in DP, you must split its defining condition into two independent halves. 

    \textcolor{gray}{For example: ``exactly one solution'' becomes ``there is one'' and ``there are no two''. ``The optimum is exactly $k$'' becomes ``at least $k$'' and ``not at least $k + 1$'' (minimizing: ``at most $k$'' and ``not at most $k - 1$'').}
    \begin{enumerate}
        \item \textbf{define the two halves}: explicitly define language $L_1$ (the ``at least'' or existence condition) and language $L_2$ (the ``not more than'' or uniqueness condition)
        \item \textbf{show the intersection}: formally state that a string $x \in L$ if and only if $x \in L_1 \cap L_2$
        \item \textbf{prove $L_1 \in \text{NP}$}: define the certificate for $L_1$ (e.g., a solution of size $k$) and state that it is polynomially bounded and can be verified in polynomial time
        \item \textbf{prove $L_2 \in \text{coNP}$} (usually by proving that $\overline{L_2}$ is in NP);
    \end{enumerate}
\end{cframe}

\section{The Class PSPACE}

\begin{defframe}{PSPACE}{}
PSPACE is the class of decision problems solvable by a deterministic Turing machine using a polynomial amount of memory space.

Formally, it is defined as PSPACE $= \bigcup_{k \ge 1} \text{SPACE}(|x|^k)$. 

A fundamental property of space complexity (Savitch's Theorem) is that $\text{PSPACE} = \text{NPSPACE}$.
\end{defframe}

\begin{itemize}
    \item While $\text{PH}$ restricts problems to a constant, fixed number of quantifier alternations, PSPACE allows for an unbounded number of alternations (up to a polynomial limit). Consequently, the entire Polynomial Hierarchy is contained within PSPACE ($\text{PH} \subseteq \text{PSPACE}$).

    \item PSPACE captures the complexity of adversarial two-player games: any two-player game with board positions of polynomial size and a polynomially bounded maximum number of moves belongs in PSPACE. The nested alternating quantifiers ($\exists \forall \exists \forall \dots$) model the ``if I move here, and you move there...'' search required to determine a guaranteed winning strategy
\end{itemize}

  \paragraph{The Complete Problem (TQBF):} the problem of determining if a True Quantified Boolean Formula (TQBF) is true is PSPACE-complete. An instance of TQBF is a fully quantified Boolean formula $Q_1 x_1 \dots Q_q x_q [\psi]$. 

\begin{itemize}
    \item unlike $\Sigma_k\text{-SAT}$, which is capped at $k$ quantifier blocks, the number of quantifiers $q$ in TQBF can grow linearly with the input size.
\end{itemize}

  TQBF can be evaluated in polynomial space using a recursive algorithm. The algorithm evaluates each quantifier $Q_i x_i$ by testing both possible values of $x_i$ (True and False). Because it can completely erase and reuse memory from the true branch when checking the false branch, the maximum memory required scales linearly with the recursion depth, keeping it in polynomial space.

\section{The Exponential Classes}
These classes scale time and space bounds to the exponential level, dealing with problems where the input size allows for massive computational resources.

\begin{defframe}{The classes}{}
    \begin{itemize}
        \item $\text{EXP} = \bigcup_{k \ge 1} \text{TIME}(2^{|x|^k})$
        \item $\text{NEXP} = \bigcup_{k \ge 1} \text{NTIME}(2^{|x|^k})$
            
    \subitem kust as $\text{NP}$ uses polynomially bounded certificates, $\text{NEXP}$ can be viewed through a certificate lens: a language $L \in \text{NEXP}$ if there are certificates of exponential size (up to $2^{|x|^c}$) that can be verified by a deterministic algorithm in time polynomial to the \textit{certificate's} size (which is exponential relative to the original input $x$)
        \item $\text{EXPSPACE} = \bigcup_{k \ge 1} \text{SPACE}(2^{|x|^k})$
    \end{itemize}
\end{defframe}

A key concept at the exponential level is how data representation affects complexity. A \textbf{succinct} representation compresses an exponentially large structure into a small input. 
\begin{itemize}
    \item \textbf{succinct graphs:} instead of listing all edges of an explicit graph, a succinct graph uses a Boolean circuit $C$, which takes two $b$-bit vertex IDs and outputs 1 if there is an edge, and 0 otherwise
    \subitem because the input $C$ is exponentially smaller than the explicit graph it represents, asking standard questions about the graph causes an exponential blowup in complexity:
    \begin{itemize}
        \item a problem in $\text{P}$ on an explicit graph becomes $\text{EXP}$-complete for a succinct graph
        \item a problem in $\text{NP}$ on an explicit graph (e.g., Hamiltonian Path) becomes $\text{NEXP}$-complete for a succinct graph
    \end{itemize}
\end{itemize}


\paragraph{Padding Arguments: Upward Translation}
Padding is a proof technique demonstrating that a collapse at a lower complexity class triggers a collapse at a higher exponential class. The core mechanism involves artificially inflating the size of an input string to shift the problem into a lower complexity class relative to its new size.

\begin{cframe}{ColorLemma}{The Padding Mechanism}
    \begin{enumerate}
        \item  Consider a language $L \in \text{NEXP}$        \item Create a new ``padded'' language $L'$ by appending dummy symbols to each input $x$ until the new length is $N = 2^{|x|^k}$
        \item  The original non-deterministic algorithm still takes $N$ steps to evaluate the unpadded prefix $x$. Because $N$ steps is polynomial relative to the massive new input size $N$, the padded language $L'$ is structurally in $\text{NP}$
        \item If we assume $\text{P} = \text{NP}$, then $L'$ must be in $\text{P}$. To decide the original $x$, a deterministic machine simply writes out the pad (taking $2^{|x|^k}$ steps) and runs the $\text{P}$ algorithm on $L'$ (taking $2^{O(|x|^k)}$ steps). This proves $L \in \text{EXP}$
    \end{enumerate}
\end{cframe}

Through padding, equalities carry upward, but inequalities carry downward via their contrapositive:
\begin{itemize}
    \item $\text{P} = \text{NP} \implies \text{EXP} = \text{NEXP}$.
    \item $\text{NP} = \text{coNP} \implies \text{NEXP} = \text{coNEXP}$.
    \item $\text{EXP} \neq \text{NEXP} \implies \text{P} \neq \text{NP}$
\end{itemize}

\chapter{A* Algorithm}

\section{Informed Search}
When trying to find the quickest or shortest route from a source node $s$ to a target node $t$ on a street network, uninformed search algorithms often aren't as fast as they could be. 

For example, Dijkstra's algorithm is a classic uninformed search algorithm (that supports non-negative weights) that always examines the closest not-yet-examined node. On a simple grid, Dijkstra's expands uniformly in all directions, corresponding essentially to a Breadth-First-Search (BFS). While accurate, its simple implementation yields a time complexity of $O(|V|^2)$, which is typically too slow for large networks.

To speed up the calculation, we can exploit the properties of the graph by applying heuristic information to evaluate each reached node and always expand the most promising one.

\subsection{Greedy Best First Search}
A purely heuristic approach evaluates nodes based on a function $h(x)$, which represents the ``direct'' (unobstructed) distance to the goal $t$. Common examples include the Euclidean distance $d_2(x,t)$ or the Manhattan distance $d_1(x,t)$. 

Greedy Best First Search always expands the reached node with the minimum distance to the goal. While this minimizes the number of expanded nodes, it \textbf{does not guarantee an optimal solution}, as it can easily get trapped by obstacles or take suboptimal detours.

\section{The A* Algorithm}

To avoid the inefficiencies of Dijkstra's algorithm and the mistakes of Greedy Best First Search, the \textbf{A* Algorithm} (Hart, Nilsson, Raphael, 1968) combines the advantages of both.

It utilizes a priority function $f(x)$ for all $x \in V$, defined as:
$$f(x) = g(x) + h(x)$$

Where:
\begin{itemize}
    \item $g(x)$ is the best known cost so far to reach node $x$ from the start node $s$ (the ``Dijkstra'' part).
    \item $h(x)$ is the estimated cost from $x$ to the goal $t$, also known as the ``cost-to-go'' (the ``Greedy'' part).
\end{itemize}

This way, $f(x)$ estimates the \textbf{total cost} of the path from $s$ to $t$ that passes through node $x$.

\begin{gframe}{Implementation and Execution Details}
During execution, A* maintains a priority queue of nodes to expand, sorted by their $f(x)$ value:

\textbf{A* Algorithm}

\begin{pseudocode}
initialize $g(s) = 0$ and $g(x) = \infty$ for all $x \in V \setminus \{s\}$
insert source $s$ into priority queue $Q$ with priority $f(s) = h(s)$
while $Q \neq \emptyset$ do
    extract node $x$ with minimum $f(x)$ from $Q$
    if $x = t$ then
        return $f(t)$ and the path (using predecessor list)
    end if
    for all outgoing edges $(x, v) \in A$ do
        $g' \gets g(x) + w_{xv}$
        if $g' < g(v)$ then // found a new or better path to $v$
            $g(v) \gets g'$
            $pred(v) \gets x$
            insert or update node $v$ in $Q$ with priority $f(v) = g' + h(v)$
        end if
    end for
end while
return no path to target $t$
\end{pseudocode}

\textbf{NB:} 
\begin{itemize}
    \item the algorithm continues until the target node $t$ is \textit{selected for expansion} (extracted from the queue), not until it is seen
    \item A* may find a shorter path to a node that has already been expanded; in this case, the node gets its $g(x)$ updated and is re-inserted into the priority queue
\end{itemize}
\end{gframe}

\section{Properties of A*}

\begin{thmframe}{Termination}{}
    A* will always terminate in a limited time.
\end{thmframe}
\begin{proofframe}
    The inner loop performs a number of iterations proportional to the out-degree of the node ($O(|V|)$), which terminates. There are finitely many acyclic paths from $s$ to other nodes in the graph, and A* only ever considers acyclic paths. On each major iteration, a new acyclic path is considered because a node is either expanded for the first time, or it is re-expanded via a shorter path. Thus, at worst, A* explores every acyclic path.
\end{proofframe}

\begin{thmframe}{Completeness}{}
    If a path from $s$ to $t$ exists, A* will always find one.
\end{thmframe}
\begin{proofframe}
    Similarly to DFS and BFS, after expanding a node $x$, A* has found paths from $s$ to all nodes directly reachable via $x$. The only case where A* stops without finding a path to $t$ is when no further reachable unexpanded nodes exist. This implies no path to $t$ exists in the graph.
\end{proofframe}

\subsection{Heuristics and Optimality}
A* is not unconditionally guaranteed to find an optimal path. Its optimality depends entirely on the properties of the chosen heuristic $h(x)$.

\begin{defframe}{Admissible Heuristic}{}
    Let $h^*(x)$ be the true minimal cost from $x$ to $t$. A heuristic $h(x)$ is \textbf{admissible} if it represents a lower bound. Formally:
    $$h(x) \le h^*(x) \quad \forall x \in V$$
    An admissible heuristic is always ``optimistic'' (it never overestimates the cost). $d_1(x,t)$, $d_2(x,t)$, and even $h(x)=0$ are all admissible heuristics.

    \textcolor{gray}{(it can be optimistic but not pessimistic because if it is optimistic you might just end up taking more time, but if it is pessimistic you might miss the shortest path (a pessimistic heuristic overestimates the cost, which can cause the algorithm to falsely assume the optimal path is too expensive and settle for a suboptimal route))}
\end{defframe}

\begin{thmframe}{Optimality}{}
    If $h(x)$ is admissible, A* is guaranteed to find a least-cost path.
\end{thmframe}
\begin{proofframe}
    Suppose A* finds a suboptimal path with cost $f' > f^* = h^*(s)$. 
    There must exist a node $x$ which is unexpanded and whose path from $s$ is the beginning of a true shortest path to $t$.
    Since A* terminated, it must be that $f(x) \ge f'$.
    However, we know:
    $f(x) = g(x) + h(x) = g^*(x) + h(x) \le g^*(x) + h^*(x) = f^*$
    So, $f^* \ge f(x) \ge f'$, which contradicts the assumption that $f' > f^*$.
\end{proofframe}

\begin{defframe}{Monotonic (Consistent) Heuristic}{}
    A heuristic $h(x)$ is \textbf{monotonic} if it satisfies the triangle inequality for all edges $(x,y) \in A$, and the goal heuristic is zero:
    $$h(x) \le w_{xy} + h(y) \quad \text{and} \quad h(t) = 0$$
    
    \begin{itemize}
        \item monotonic heuristics ensure that the $f$-values along any path are non-decreasing ($f(x) \le f(y)$)
        \item it is impossible to decrease $f(x)$ by extending a path
        \item this guarantees no re-expansions of nodes are ever necessary, making it comparable to Dijkstra (on non-negative weights)

\begin{gframe}{\textbf{N.B.}: monotonicity implies admissibility!}
\end{gframe}

    \end{itemize}
\end{defframe}

\section{Complexity and Variants}

A* is optimally efficient with respect to the used heuristic $h(x)$, meaning no other complete algorithm using the exact same heuristic will expand fewer nodes. 
\begin{itemize}
    \item note that Dijkstra's algorithm is the special case of A* where $h(x) = 0$ !
\end{itemize}

The runtime is generally $O(d^l)$ (where $l$ is the number of nodes on the shortest path and $d$ is a constant), but the actual performance relies heavily on the quality of $h(x)$.

\begin{gframe}{Heuristic Dominance and the 8-Puzzle}
    A better heuristic leads to better A* performance. If $h_1(x) \le h_2(x)$ for all $x \in V$, we say that $h_2$ \textbf{dominates} $h_1$. A* is guaranteed to expand no more nodes using $h_2$ than it would with $h_1$.

    With a nearly perfect heuristic $h(x) \approx h^*(x)$, runtime drops to $\Theta(l)$.
    
    For example, for the 8-Puzzle problem:
    \begin{itemize}
        \item $h_1(x)$: number of tiles in wrong positions.
        \item $h_2(x)$: sum of Manhattan distances of each tile to its target location.
    \end{itemize}
    Because $h_2$ dominates $h_1$, running A* with $h_2$ reduces the number of expanded nodes compared to $h_1$ (e.g., 1,641 nodes vs 39,135 nodes for a 24-move solution).
\end{gframe}

\subsection{Improvements and Variants}
For very complex problems, A* might still consume too much memory or time. Common improvements include:

\begin{itemize}
    \item \textbf{Bidirectional search}: running search from $s$ to $t$ and from $t$ to $s$ simultaneously until they meet.
    \item \textbf{Weighted A*}: ``inflating'' the heuristic (e.g., $f(x) = g(x) + W \cdot h(x)$) to place more emphasis on reaching the goal quickly.
        \subitem this sacrifices admissibility (optimality is no longer guaranteed), but it provides a bounded sub-optimality in exchange for big speedups
    \item \textbf{Preprocessing}: pre-calculating all-pairs shortest paths or storing pairwise minimum distances to create near-perfect heuristics
    \item \textbf{Hierarchical approach}: searching on abstraction levels (e.g., plan a route on major highways first, then refine to local main roads, and finally down to street level)
    \item \textbf{Iterated Deepening A* (IDA*)}: designed primarily to limit memory usage; it mimics Iterative Deepening DFS, performing a depth-first search while pruning any node where $f(x)$ exceeds a progressively increasing threshold $c$. 
        \subitem while it guarantees finding the optimal solution with lower memory bounds, it is generally much slower than standard A* due to repeated work
\end{itemize}

\chapter{Planar Graphs}

\section{Basic Concept}
A graph $G=(V,E)$ is called planar if it can be drawn in the plane in a way such that no two edges intersect geometrically except at a vertex to which both are incident. Such a drawing is called a \textbf{planar embedding} of $G$, or a plane graph. A plane graph $G$ divides the plane into connected regions called \textbf{faces}. The unbounded region is called the outer or external face of $G$.

\section{Euler's Formula}
\begin{thmframe}{Euler's formula}{}
Let $G=(V,E)$ be a connected, planar graph, and let $f$ denote the number of faces. Then 
$$|V|-|E|+f=2$$
holds for any planar embedding of $G$.
\end{thmframe}

From Euler's formula, we can derive several important corollaries:
\begin{itemize}
    \item if $G$ is a simple, planar graph with $|V| \ge 3$, then the number of edges is bounded by $|E| \le 3|V| - 6$
    \item for any simple, planar, bipartite graph with $|V| \ge 3$ vertices and $|E|$ edges, the bound is tighter: $|E| \le 2|V| - 4$
    \item each simple, planar graph contains a vertex $v$ of degree $d(v) \le 5$
\end{itemize}

\section{Linear Planarity Testing Algorithms}
Several algorithmic approaches have been developed to test planarity and construct planar embeddings in linear time:

\begin{itemize}
    \item \textbf{Path-Addition Approach}: the first linear-time planarity testing algorithm, introduced by Hopcroft and Tarjan (1974). 
        \subitem starts from a cycle and adds one path at a time to the structure
        \subitem \textcolor{gray}{(Mehlhorn and Mutzel (1996) later clarified how to use this to construct the actual planar embedding)}
    \item \textbf{Vertex-Addition Approach}: first considered by Lempel, Even, and Cederbaum (1967)
        \subitem  adds vertices one-by-one following an $st$-numbering
        \subitem \textcolor{gray}{(Booth and Lueker (1976) proved it can be implemented in linear time using PQ-trees, and Chiba et al. (1985) showed how to construct the embedding)}
    \item \textbf{Block Embedding Approach}: proposed by Shih and Hsu (1999) and Boyer and Myrvold (1999)
        \subitem first constructs a DFS-tree and then adds back edges to iteratively construct a planar embedding if one exists
        \subitem \textcolor{gray}{Boyer and Myrvold (2004) later introduced a new variant with a simpler implementation based on edge addition}
\end{itemize}

\chapter{Randomized Algorithms}

\section{Some Probability Theory}
\subsection{The Basics}
\begin{defframe}{Finite Probability Space}{}
    \begin{itemize}
        \item \textbf{sample space $\Omega$}: set of all possible outcomes (e.g., all possible outcomes of rolling a dice)
            \subitem every point $i \in \Omega$ has a nonnegative probability $p(i)$, such that $\sum_{i \in \Omega} p(i) = 1$
        \item event $\mathcal{E}$: subset of the sample space ($\mathcal{E} \subseteq \Omega$).
        \item \textbf{probability of $\mathcal{E}$}: 
            $$Pr[\mathcal{E}] = \sum_{i \in \mathcal{E}} p(i)$$
        \item complement: 
            $$Not(\mathcal{E}) = \Omega - \mathcal{E}$$
    \end{itemize}
\end{defframe}

\begin{defframe}{Conditional probability \& independence}{}
    \begin{itemize}
        \item \textbf{conditional probability} of event $\mathcal{E}$ given event $\mathcal{F}$:
        $$Pr[\mathcal{E} | \mathcal{F}] = \frac{Pr[\mathcal{E} \cap \mathcal{F}]}{Pr[\mathcal{F}]}$$
        \item \textbf{independence}: 

            events $\mathcal{E}_1, \dots, \mathcal{E}_n$ are independent if and only if for all subsets $I \subseteq \{1, \dots, n\}$:
        $$Pr\left[\bigcap_{i \in I} \mathcal{E}_i\right] = \prod_{i \in I} Pr[\mathcal{E}_i]$$
    \end{itemize}
\end{defframe}

\begin{cframe}{ColorThm}{Union bound}{}
    For any events $\mathcal{E}_1, \dots, \mathcal{E}_n$, the probability of their union is bounded by the sum of their individual probabilities:
    $$Pr\left[\bigcup_{i=1}^n \mathcal{E}_i\right] \le \sum_{i=1}^n Pr[\mathcal{E}_i]$$
\end{cframe}

\subsection{Random Variables and Expected Values}

A \textbf{random variable} $X: \Omega \rightarrow \mathbb{N}$ is a function from the sample space to the natural numbers. 

The \textbf{expected value} (or expectation) of $X$ is defined as:
$$E[X] = \sum_{j=0}^\infty j \cdot Pr[X = j]$$

\textbf{example: waiting for a first success} \\
Suppose a trial is successful with probability $0 < p < 1$ and fails with probability $1 - p$. If $X$ is the number of independent trials until the first success, the expectation is:
$$E[X] = \sum_{j=0}^\infty j \cdot (1-p)^{j-1}p = \frac{p}{1-p} \sum_{j=0}^\infty j \cdot (1-p)^j = \frac{p}{1-p} \cdot \frac{1-p}{p^2} = \frac{1}{p}$$

\begin{cframe}{ColorThm}{Linearity of expectation}{}
    Let $X$ and $Y$ be random variables over the same probability space. Then:
    $$E[X + Y] = E[X] + E[Y]$$
\end{cframe}

\begin{gframe}{Example: guessing cards}
Given a deck of $n$ cards, repeatedly guess the top card before turning it over. Let $X$ be the number of correctly guessed cards, and $X_i = 1$ if the $i$-th card is guessed correctly (0 otherwise).
\begin{itemize}
    \item \textit{memoryless}: 
        $$E[X_i] = \frac{1}{n} \ \ \ \text{ so } \ \ \ E[X] = \sum_{i=1}^n \frac{1}{n} = 1$$
    \item \textit{with memory}: 
        $$E[X_i] = \frac{1}{n-i+1} \  \ \ \text{ so } \  \ \ E[X] = \sum_{i=1}^n \frac{1}{n-i+1} = \sum_{i=1}^n \frac{1}{i} = \mathcal{H}_n \approx \ln n$$ 
    (where $\mathcal{H}_n$ is the $n$-th Harmonic number, satisfying $\ln(n+1) < \mathcal{H}_n \le 1 + \ln n$).
\end{itemize}
\end{gframe}

\begin{gframe}{Example: coupon collector's problem}
Assume there are $n$ types of coupons, and in each round, you get one at random (equally likely). 

Let $X$ be the number of rounds to collect all types. 

We define \textit{phase $j$} as the period where you have $j$ types and are waiting for the $(j+1)$-th type. Let $X_j$ be the number of rounds in phase $j$.
\begin{itemize}
    \item the success probability in phase $j$ is $p(j) = \frac{n-j}{n}$.
    \item the expected rounds for phase $j$ is $E[X_j] = \frac{1}{p(j)} = \frac{n}{n-j}$.
    \item by linearity of expectation: $E[X] = \sum_{j=0}^{n-1} E[X_j] = n \sum_{j=0}^{n-1} \frac{1}{n-j} = n \, \mathcal{H}_n \approx n \ln n$.
\end{itemize}
\end{gframe}

\section{Randomized Algorithms}
Randomized algorithms depend on \textbf{uniformly random numbers} as an \textbf{auxiliary input} to guide behaviour (these numbers are usually implemented by a pseudo-random number generator).

\begin{gframe}{Example: quicksort}
Deterministic quicksort has a worst-case runtime of $\Theta(n^2)$, when it gets handed specific input permutations (e.g., an already sorted array). 

We can randomise quicksort by selecting the pivot element uniformly at random instead of using a deterministic position(standard quicksort has a ``fixed'' pivot).

This way, the worst case is still $\Theta(n^2)$, but it happens only if the internal random number generator repeatedly picks extreme values (very unlikely) (so, since choices are random, there are no bad input permutations that will always run in $\Theta(n^{2})$).

The expected runtime is $\Theta(n \log n)$ for all possible input permutations. The expectation is taken over all possible random decisions made by the algorithm, avoiding dependency on the input data distribution.
\end{gframe}


Randomized algorithms reduce runtime and/or memory usage for worst case input, and they avoid dependency on input data.

\begin{defframe}{Monte Carlo vs Las Vegas}{}
    There are two types of randomized algorithms:
    \begin{itemize}
        \item \textbf{Monte Carlo} algorithms: always \textbf{time-efficient}, but not always correct (correct with high probability)
        \item \textbf{Las Vegas} algorithms: always \textbf{correct}, but not always time-efficient
    \end{itemize}
\end{defframe}

\subsection{Randomized Primality Test}
We are given a positive integer $n$, and we want to determine whether it is prime.

The first issue we run into is "bit complexity". Standard algorithm analysis often assumes arithmetic operations (like addition) take constant time ($O(1)$). However, this is only true for numbers small enough to fit in standard CPU registers (e.g., 64-bit integers). 

    For arbitrarily large numbers, addition takes time proportional to the length of the number. Therefore, we use \textbf{bit complexity}:
    \begin{itemize}
        \item represent $n$ as a binary number with $k = \lceil \log_2 n \rceil + 1$ bits
        \item the number of fundamental bit operations (rather than high-level arithmetic operations) becomes the criterion for time complexity
    \end{itemize}

    \textcolor{gray}{For example, if we wanted to check primality via the "division method" - checking if $n$ is divisible by $2$ or by some odd number in $\{3, \dots, \lfloor\sqrt{n}\rfloor \}$ -, we would need to do $O(\sqrt{n})$ divisions, with a bit complexity of $\frac{O(2^{k}{2}})$ with $k = \Theta(\log n)$}.

We need to use another method. We can make use of \textbf{Fermat's Little Theorem}, which states:
$$\text{if } n \text{ is prime, } a^{n-1}\equiv 1 \text{ mod } n \ \text{for all } a \in \{1, \dots, n-1\}$$

This allows us to say that, if for some $a$
\begin{itemize}
    \item $a^{n-1} \not \equiv 1 \text{ mod } n$, then $n$ is definitely not prime (and $a$ is a witness for $n$'s compositeness)
    \item $a^{n-1} \equiv 1 \text{ mod } n$ then $n$ is prime with some probability
\end{itemize}

\vspace{1em}
This gives us the following algorithm:
\vspace{0.5em}

\begin{pseudocode}
procedure FermatPrimalityTest($n, s$)
    for $i \gets 1$ to $s$ do
        $a \gets \text{random}(2, n - 1)$
        if Witness($a, n$) then
            return COMPOSITE
        end if
    end for
    return PRIME
end procedure
\end{pseudocode}

Which essentially says that, if no witness for the number has been found after $s$ iterations, then $n$ is likely prime.
\begin{itemize}
    \item this is a \textbf{Monte Carlo} algorithm (as it could return a false positive)
\end{itemize}

To implement this efficiently, we need a way to compute $a^{n-1} \bmod n$ in a cheap way.

\begin{gframe}{Efficient Modular Exponentiation}
We can break down the exponentiation using the binary representation of $n-1$: $(b_k, b_{k-1}, \dots, b_1)$. 

We then have that:
\begin{itemize}
    \item $a^{2c} \bmod n = (a^c \bmod n)^2 \bmod n$
    \item $a^{c+1} \bmod n = (a^c \bmod n) \cdot a \bmod n$
\end{itemize}
\end{gframe}

This yields the following algorithm for the witness check:

\vspace{0.5em}
\begin{pseudocode}
procedure Witness($a, n$)
    Let $(b_k, \dots, b_1)$ be the binary representation of $n-1$
    $result \gets 1$
    for $i \gets k$ down to $1$ do
        $result \gets (result \cdot result) \bmod n$
        if $b_i = 1$ then
            $result \gets (result \cdot a) \bmod n$
        end if
    end for
    if $result \neq 1$ then
        return true // a is a witness for n not being prime
    else
        return false // a is not a witness
    end if
end procedure
\end{pseudocode}


\begin{itemize}
    \item multiplication and modulo of two $k$-bit numbers takes $O(k^2)$ time
    \item the "witness" function runs in $O(k^3)$
    \item the total running time of the Fermat algorithm is $O(s \cdot k^3)$
\end{itemize}

The test fails for \textit{Carmichael numbers}, which are composite but incorrectly satisfy $a^{n-1} \equiv 1 \bmod n$ for all $a \in \{1, \dots, n-1\}$. Fortunately, these are extremely rare (e.g., there are only 246,683 below $10^{16}$).

For all other odd, non-prime numbers $n > 2$:
\begin{itemize}
    \item there are at least $\frac{n-1}{2}$ witnesses
    \item in each iteration, a randomly chosen $a$ is a witness with probability $\ge \frac{1}{2}$
    \item after $s$ iterations, the probability of not finding a witness is $\le \frac{1}{2^s}$
\end{itemize}

\section{Analysis of (Randomized) Quicksort}
The dominant cost of the algorithm is the comparison of elements to the pivot.

To analyse this, we conceptually sort the elements of the array as $x_{1} < x_{2} < \dots < x_{n}$, and we define a variable $X_{i,j}$ s.t. 
 
    $$X_{i,j} = \begin{cases} 1 & \text{if } x_i \text{ is compared to } x_j \\ 0 & \text{otherwise} \end{cases}$$
    
Because the pivot is selected uniformly at random from the current subarray $A[l \dots r]$, we have:
    $$Pr[\text{random}(l,r) = i] = \frac{1}{r-l+1}$$

We can visualise the recursive pivot choices as a \textbf{binary search tree} (where the nodes are the randomly chosen pivot elements), in which every element is only compared with its direct ancestors or descendants.

Very importantly, if we consider two elements $x_{i}$ and $x_{j}$, we know that they will only ever be compared if and only if \textit{one of them is chosen as a pivot} before any element in between them is chosen: if any intermediate element $x_k$ (where $i < k < j$) is selected as a pivot first, it would split them: $x_i$ goes to the left subarray, $x_j$ goes to the right subarray, and they would never be compared.

\begin{cframe}{ColorLemma}{Probability of a single comparison}
    Consider the subset of elements $\{x_i, x_{i+1}, \dots, x_j\}$. This subset contains $j - i + 1$ elements. 
    
    Since pivots are chosen uniformly at random, every element in this subset is equally likely to be the first one chosen as a pivot. The only choices that result in a comparison between $x_i$ and $x_j$ are picking $x_i$ or picking $x_j$. 
    
    Therefore, the probability that $x_i$ and $x_j$ are compared is exactly:
    $$Pr[X_{i,j}] = \frac{2}{j - i + 1}$$
\end{cframe}

\begin{cframe}{ColorThm}{Expected total comparisons}
    We want to find the expected total number of comparisons, $E[C]$. 

    Using the Linearity of Expectation, we sum the expectations (which equal the probabilities) of all indicator variables:
    
    $$E[C] = E\left[\sum_{i=1}^{n-1} \sum_{j=i+1}^n X_{i,j}\right] = \sum_{i=1}^{n-1} \sum_{j=i+1}^n Pr[X_{i,j}] = \sum_{i=1}^{n-1} \sum_{j=i+1}^n \frac{2}{j - i + 1}$$
    
    We simplify this using a change of variables, letting $k = j - i + 1$:
    $$E[C] = \sum_{i=1}^{n-1} \sum_{k=2}^{n-i+1} \frac{2}{k} \le 2 \sum_{i=1}^n \sum_{k=1}^n \frac{1}{k} = 2n\mathcal{H}_n$$
    
    Where $\mathcal{H}_n = \sum_{k=1}^n \frac{1}{k}$ is the $n$-th Harmonic number. Since $\ln(n+1) < \mathcal{H}_n \le 1 + \ln n$, we know $\mathcal{H}_n \approx \ln n$.
    
    \textbf{Therefore:} 
    $$E[C] = E[T] = \Theta(n \log n)$$
\end{cframe}


\section{MAX 3-SAT}

\begin{defframe}{MAX-3SAT Problem}{}
    \textbf{input:} a set of clauses $C_1, \dots, C_k$, each of length 3, over a set of binary variables $X = \{x_1, \dots, x_n\}$.
    
    \textbf{goal:} find a variable assignment satisfying as many clauses as possible. 
    
    \begin{gframe}{MAX-3SAT is NP-hard!}
    \end{gframe}
\end{defframe}

\textbf{Randomized idea:} set each variable independently to \text{true} with probability $\frac{1}{2}$ and to \text{false} otherwise.

\begin{cframe}{ColorThm}{Expected satisfied clauses}
    Given a MAX-3SAT instance with $k$ clauses, the expected number of clauses satisfied by a random assignment is $\frac{7}{8}k$.
\end{cframe}
    
\begin{proofframe}{}{}
    Define an indicator variable $Z_j$ which equals 1 if clause $C_j$ is satisfied, and 0 otherwise. 
    Let $Z = \sum_{j=1}^k Z_j$ be the total number of satisfied clauses.
    
    A clause of length 3 fails only if all 3 literals are false. Since each variable is chosen independently with probability $\frac{1}{2}$, the probability a clause fails is $\left(\frac{1}{2}\right)^3 = \frac{1}{8}$. 

    Thus, $Pr[C_j \text{ is satisfied}] = 1 - \frac{1}{8} = \frac{7}{8}$.
    
    By linearity of expectation:
    $$E[Z] = \sum_{j=1}^k E[Z_j] = \sum_{j=1}^k Pr[C_j \text{ is satisfied}] = \frac{7}{8}k$$
\end{proofframe}
\begin{cframe}{ColorLemma}{Lower Bound on Satisfiable Clauses}
    For \textit{every} instance of MAX-3SAT there is an assignment that satisfies $\ge \frac{7}{8}$ of all clauses.
    
\end{cframe}

\begin{proofframe}{}{}
As $E[Z] = \frac{7}{8}k$, the probability $Pr[Z \ge \frac{7}{8}k]$ of constructing such an assignment must be greater than 0. Consequently, such an assignment must exist. 
\begin{gframe}{}
     We can show the existence of a structure by providing a random construction process that succeeds with positive probability.
\end{gframe}

\end{proofframe}

\subsection{Johnson's Algorithm}

While the expectation is $\frac{7}{8}k$, a random variable might almost always be below its mean. 

To have a reliable algorithm, we must determine the actual probability $p$ of generating a good assignment.

\begin{cframe}{ColorLemma}{Probability of Success}
    The probability $p$ that a random assignment satisfies $\ge \frac{7}{8}k$ clauses is $\ge \frac{1}{8k}$.
\end{cframe}
    
\begin{proofframe}{}{}
Let $p_j$ be the probability that exactly $j$ clauses are satisfied. 

Then $p = \sum_{j \ge 7k/8} p_j$.
    $$\frac{7}{8}k = E[Z] = \sum_{j < 7k/8} j \cdot p_j + \sum_{j \ge 7k/8} j \cdot p_j$$
    $$\le \frac{7k-1}{8} \sum_{j < 7k/8} p_j + k \sum_{j \ge 7k/8} p_j = \frac{7k-1}{8}(1-p) + kp \le \frac{7k-1}{8} + kp$$
    $$kp \ge \frac{7}{8}k - \frac{7k-1}{8} = \frac{1}{8} \implies p \ge \frac{1}{8k}$$
\end{proofframe}

The algorithm used to solve MAX-3SAT with this probability is called \textbf{Johnson's algorithm}.

    \textbf{the algorithm:} repeatedly generate random assignments until one satisfies $\ge \frac{7}{8}k$ clauses.
    \begin{itemize}
        \item it's a $7/8$-approximation algorithm for MAX-3SAT. \textcolor{gray}{(N.B.: Håstad proved that no $\alpha$-approximation exists for $\alpha > 7/8$ unless P=NP).}
        \item as seen above, each iteration succeeds with probability $\ge \frac{1}{8k}$; using the waiting-time bound for a first success, the expected number of trials is $\le 8k$.
        \item the total expected runtime is $E[T] = O(8k) \cdot O(n+k) = O(k^2 + kn)$
    \end{itemize}
    
\begin{gframe}{}
    This process turns a Monte Carlo algorithm into a Las Vegas algorithm (it always produces a correct $7/8$ approximation, but runtime is an expectation).
\end{gframe}

\section{Tail Inequalities}

Expected runtimes describe average behavior, but tail inequalities provide bounds on the probability that an algorithm's performance strays far from its expectation.

\begin{cframe}{ColorThm}{Markov's Inequality}{}
    Let $X$ be a non-negative random variable ($X \ge 0$) with expectation $E[X]$.

    For any $t > 0$:
    $$Pr[X \ge t] \le \frac{E[X]}{t} \quad \text{or with } t = k \cdot E[X]: \quad Pr[X \ge k \cdot E[X]] \le \frac{1}{k}$$
   
\begin{itemize}
    \item requires the least information (only $E[X]$ and non-negativity), making it universally applicable but providing the weakest (slowest decay) bound
\end{itemize}
\end{cframe}

\begin{cframe}{ColorThm}{Chebyshev's Inequality}{}
    Let $X$ be a random variable with expectation $E[X]$ and standard deviation $\sigma[X] = \sqrt{E[(X - E[X])^2]}$. For any $t > 0$:
    $$Pr[|X - E[X]| \ge t \cdot \sigma[X]] \le \frac{1}{t^2}$$
    
    \begin{itemize}
        \item incorporates variance to measure spread around the mean; the deviation probability drops quadratically ($1/t^2$)
    \end{itemize}
\end{cframe}

\begin{cframe}{ColorThm}{Chernoff Bounds}{}
    Let $X = \sum_{i=1}^n X_i$ be the sum of \textit{independent 0--1 random variables}. 

    For any $\mu \ge E[X]$:
    
    \textbf{above the mean (for any $\delta > 0$):}
    $$Pr[X > (1+\delta)\mu] < \left(\frac{e^\delta}{(1+\delta)^{1+\delta}}\right)^\mu$$
    
    \textbf{below the mean (for $0 < \delta < 1$):}
    $$Pr[X < (1-\delta)\mu] < e^{-\delta^2 \mu / 2}$$
    
    \begin{itemize}
        \item the sum of independent 0-1 random variables is exponentially tightly concentrated around its mean
    \end{itemize}
\end{cframe}
\newpage
\textbf{Example: Chernoff bounds on coin flips}\\
Consider $n$ flips of a fair coin, where $X$ is the total number of heads. 

Here, $\mu = E[X] = \frac{n}{2}$. 

We want the probability of getting more than $\frac{3}{4}n$ heads (i.e., more than $75\%$ heads).

We set the Chernoff threshold $(1 + \delta)\mu$ equal to $\frac{3}{4}n$:
$$(1 + \delta) \cdot E[X] = \frac{3}{4}n \iff (1 + \delta)\left(\frac{n}{2}\right) = \frac{3}{4}n$$
$$1 + \delta = \frac{3}{2} = 1.5 \implies \delta = 0.5$$

So, we have that
$$Pr\left[X > (1 + 0.5)E[X]\right] = Pr\left[X > \frac{3}{4}n\right] < \left( \frac{\sqrt{e}}{1.5^{1.5}} \right)^{n/2} \approx 0.9^{n/2}$$

If we compare the results with the actual probabilities:
\begin{itemize}
    \item $n = 2$: exact $Pr\left[X > \frac{3}{4}n\right] = 0.25 < 0.9$
    \item $n = 3$: exact $Pr\left[X > \frac{3}{4}n\right] = 0.125 < 0.86$
    \item $n = 4$: exact $Pr\left[X > \frac{3}{4}n\right] = 0.0625 < 0.81$
    \item $n = 10$: exact $Pr\left[X > \frac{3}{4}n\right] = 0.0546875 < 0.59$
    \item $n = 100$: exact $Pr\left[X > \frac{3}{4}n\right] \approx 2.82 \cdot 10^{-7} < 0.0052$
\end{itemize}

Chernoff bounds overestimate the true error probability significantly, but they are widely used because they are simple to compute and guarantee exponential decay wrt $n$.
\section{Key Takeaways}

\begin{cframe}{ColorThm}{The Probabilistic Method}
Non-constructive proof technique used to prove the \textbf{existence of a specific solution or structure}. It can be done:
\begin{itemize}
    \item \textbf{via expectation:} if you can show that the expected value of a random variable is $E[X] = \mu$, then there mathematically must exist at least one specific outcome in the sample space that achieves $\ge \mu$ (and analogously, one that achieves $\le \mu$)
    \item \textbf{via probability:} if you can prove that the probability of randomly generating a valid configuration is greater than zero ($Pr[\mathcal{E}] > 0$), then that configuration definitively exists
\end{itemize}
\end{cframe}

\begin{cframe}{ColorLemma}{Converting Monte Carlo to Las Vegas}
A Monte Carlo algorithm (fast, but potentially returns an incorrect output) can be upgraded into a Las Vegas algorithm (always correct, but with an expected, non-guaranteed runtime) if and only if you can efficiently verify the correctness of the output:
\begin{itemize}
    \item wrap the Monte Carlo algorithm in a loop that repeats until the verification step succeeds
\end{itemize}
\end{cframe}

\begin{cframe}{ColorDef}{Analyzing Algorithm Probability}{}
To analyze the expected runtime or an expected score of a randomized algorithm:
\begin{enumerate}
    \item \textbf{define indicator variables:} break the total cost down into binary events. 
       \subitem define an indicator variable $X_i \in \{0, 1\}$ \textcolor{gray}{(e.g., $X_{i,j} = 1$ if a comparison happens between $i$ and $j$, or $Z_j = 1$ if clause $j$ is satisfied).}
    \item \textbf{calculate individual probability:} compute the probability that a single indicator evaluates to 1 ($Pr[X_i = 1]$)
        \subitem because it is a binary 0-1 variable, its expected value is exactly its probability: $E[X_i] = Pr[X_i = 1]$
    \item \textbf{apply linearity of expectation:} use $E[\sum X_i] = \sum E[X_i]$ to add up all the individual probabilities, getting the overall expected total cost
    \end{enumerate}
    \end{cframe}

\begin{cframe}{ColorThm}{Choosing a Tail Inequality}
Tail inequalities quantify concentration by bounding the probability that a random variable deviates far from its mean. We can choose the bound based on what information is available:
\begin{itemize}
    \item \textbf{markov's inequality:} use when $X \ge 0$ and you \textit{only} know $E[X]$.
        \subitem weakest bound; probability drops off polynomially ($O(1/k)$).
    \item \textbf{chebyshev's inequality:} use when you know both $E[X]$ and the variance $\sigma^2[X]$ (or standard deviation $\sigma$).
        \subitem wtronger bound; probability drops off quadratically ($O(1/t^2)$).
    \item \textbf{chernoff bounds:} use when $X = \sum X_i$ is a sum of \textit{independent 0--1} (Bernoulli) random variables.
        \subitem wtrongest bound; probability drops off exponentially.
\end{itemize}
\end{cframe}

\begin{cframe}{ColorDef}{Summary of Tail Inequalities}{}
    \vspace{-1em}
\begin{enumerate}
    \item \textbf{Markov:} $Pr[X \ge k \cdot E[X]] \le \frac{1}{k}$
        \subitem \textit{assumptions:} $X \ge 0$, $k > 0$.
    \item \textbf{Chebyshev:} $Pr[|X - E[X]| \ge t \cdot \sigma[X]] \le \frac{1}{t^2}$
        \subitem \textit{assumptions:} Known variance $\sigma^2[X]$.
    \item \textbf{Chernoff (Above Mean):} $Pr[X > (1+\delta)\mu] < \left( \frac{e^\delta}{(1+\delta)^{1+\delta}} \right)^\mu$
        \subitem \textit{assumptions:} $X = \sum_{i=1}^n X_i$ with independent $X_i \in \{0, 1\}$ and $\mu \ge E[X]$.
    \item \textbf{Chernoff (Below Mean):} $Pr[X < (1-\delta)\mu] < e^{-\delta^2 \mu / 2}$
        \subitem \textit{Assumptions:} Same as above, with $0 < \delta < 1$.
\end{enumerate}
\end{cframe}
